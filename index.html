    base_url="https://shelf-plants-shark-thunder.trycloudflare.com/v1",

# Supported vision models: "fara1.5", "qwen3.8-27b-uncensored"

    model="fara1.5",
    messages=[{
        "role": "user",
        "content": [
            {"type": "text", "text": "Describe the contents of this screenshot."},
            {"type": "image_url", "image_url": {"url": "data:image/png;base64,..."}}
        ]
    }]
)
print(response.choices[0].message.content)
